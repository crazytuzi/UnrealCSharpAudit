```cpp
FString UDynamicDataSource::GetVirtualPath(const UClass* InClass)
{
	FString VirtualPath;

	if (InClass != nullptr)
	{
		const auto InternalPath = DYNAMIC_ROOT_INTERNAL_PATH / InClass->GetFName().ToString();

		FName ConvertedVirtualPath;

		if (TryConvertInternalPathToVirtual(FName(*InternalPath), ConvertedVirtualPath))
		{
			VirtualPath = ConvertedVirtualPath.ToString();
		}
	}

	return VirtualPath;
}
```
